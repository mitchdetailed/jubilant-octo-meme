# esses_formats (layer 2, Qt-free): importers, readers and writers (05_IMPORTERS_AND_PLUGINS.md).
#
# M0: the .ld/.ldx binary core in src/ld/ (namespace esses::formats::ld, internal), and on top of it the
# importer layer of 05 sections 1-2 (ByteSource, ProbeResult/probe_json, IImporter/IReader/IWriter,
# ImporterRegistry and the open pipeline into esses_model) with the motec.ld importer and writer in
# src/importer/. src/ld/ keeps including nothing above esses_store (plus pugixml for .ldx); src/importer/ and
# the files directly under src/ bridge it to esses_units and esses_model (08 section 2 table). Floating-point
# text is parsed with fast_float everywhere in this library, never std::from_chars, because Apple Clang 15's
# libc++ (the macos-14 runner) has no floating-point from_chars.
#
# M2: library sources are globbed (CONFIGURE_DEPENDS) so parallel work adds files without editing this file; the
# text importers (TextEngine, TextProfile, Dictionary, UnitMappingProfile, the CSV readers and writers) live in
# src/text/ and their tests in tests/text/ (one Catch2 executable, formats_text.<test>). The two .ld test
# executables keep their explicit lists.

# Writer identity (05 section 5 item 1): Esses' own neutral constants, derived from cmake/naming.cmake so a
# rename touches one place.
set(ESSES_FORMATS_LD_DEVICE_TYPE "${ESSES_APP_NAME}")
configure_file(src/formats_build_config.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/formats_build_config.hpp"
               @ONLY)

# M2: the shipped text profiles and their dictionaries (data/profiles-text, data/dictionaries) embedded as
# string constants, so ImporterRegistry::withBuiltins registers motec.csv and generic.csv without a data
# directory (the CLI, the fuzz targets, the tests). The data files stay the one source; editing one re-runs
# the configure step.
set(_esses_text_builtins_content
    "// Generated by libs/formats/CMakeLists.txt from data/profiles-text and data/dictionaries; do not edit.\n")
string(APPEND _esses_text_builtins_content
       "#pragma once\n#include <string_view>\nnamespace esses::formats::text::builtin {\n")
foreach(_pair "kLegacyCsvProfile=profiles-text/motec.csv.json" "kLegacyCsvDictionary=dictionaries/motec.csv.json"
              "kGenericCsvProfile=profiles-text/generic.csv.json" "kGenericCsvDictionary=dictionaries/generic.csv.json")
  string(REPLACE "=" ";" _parts "${_pair}")
  list(GET _parts 0 _name)
  list(GET _parts 1 _file)
  set(_path "${CMAKE_SOURCE_DIR}/data/${_file}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_path}")
  file(READ "${_path}" _json)
  string(APPEND _esses_text_builtins_content
         "inline constexpr std::string_view ${_name} = R\"esses_json(${_json})esses_json\";\n")
endforeach()
string(APPEND _esses_text_builtins_content "} // namespace esses::formats::text::builtin\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/builtin_text_profiles.hpp.tmp" "${_esses_text_builtins_content}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/generated/builtin_text_profiles.hpp.tmp"
               "${CMAKE_CURRENT_BINARY_DIR}/generated/builtin_text_profiles.hpp" COPYONLY)

file(GLOB_RECURSE _esses_formats_sources CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp"
     "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp")
esses_add_library(formats LAYER 2 QT_FREE
  SOURCES ${_esses_formats_sources}
  DEPS
    esses::base
    esses::units
    esses::store
    esses::model
  PRIVATE_DEPS
    esses::pugixml
    esses::nlohmann_json
    esses::fast_float)
target_include_directories(${ESSES_PREFIX}_formats PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
